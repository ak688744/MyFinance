# L4.1 Investment Analyzer Agent (v1: Portfolio Health) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Investment Analyzer specialist agent — store fund fundamentals + holdings for owned MF, run deterministic health analyses (overlap, look-through allocation, concentration, redundancy, cost, goal-fit), expose them via MCP read tools + REST, let an isolated agent refresh fund data itself and reason over verified numbers, and surface deterministic MF insight cards + a Discuss chat drawer on the Investments page.

**Architecture:** Additive-only. New `core` tables (`scheme_fundamentals`, `scheme_holdings`) reference `investment_schemes`; MF XIRR/portfolio/NAV math is untouched (handed in as ground truth, never recomputed). Ingestion is deterministic (structured-JSON fetch adapters + Zod, no LLM/gateway). Analysis functions are pure `core/domain` (textbook-tested). The agent mirrors `buildExpenseAgent` (isolated memory resource, focused tool allowlist). No new stream contract — reuse the existing `POST /agent/chat` SSE and `agent` field.

**Tech Stack:** TypeScript (Node 22), Drizzle + better-sqlite3 (synchronous repos), Zod, Fastify, `@modelcontextprotocol/sdk`, Mastra (`@mastra/core`), Vite + React 19 + TanStack Query, vitest.

**Spec:** `docs/superpowers/specs/2026-09-06-l4.1-investment-analyzer-agent-design.md` (read it alongside this plan — the plan argues from the spec's D1–D9 decisions).

import type { Db } from '../db/client';
import { schemeFundamentals } from '../db/schema';
import type { SchemeFundamentals } from '../types';
import type { SchemeFundamentalsRepo } from './types';

export function makeSchemeFundamentalsRepo(db: Db): SchemeFundamentalsRepo {

    upsert(row) {
      const values = {
        schemeId: row.schemeId,
        expenseRatioDirect: row.expenseRatioDirect,
        expenseRatioRegular: row.expenseRatioRegular,
        planType: row.planType,
        aum: row.aum,

        source: row.source,
        ...(row.fetchedAt ? { fetchedAt: row.fetchedAt } : {}),

      db.insert(schemeFundamentals).values(values)
        .onConflictDoUpdate({
          target: schemeFundamentals.schemeId,
          set: { ...values, fetchedAt: row.fetchedAt ?? sql`CURRENT_TIMESTAMP` },

        .run();
    },
    get(schemeId) {

      return (r ?? null) as SchemeFundamentals | null;
    },
  };
}

```ts
import { and, desc, eq, notInArray, sql } from 'drizzle-orm';
import type { Db } from '../db/client';
import { schemeHoldings } from '../db/schema';
import type { StoredFundHolding } from '../types';
import type { SchemeHoldingsRepo } from './types';

export function makeSchemeHoldingsRepo(db: Db): SchemeHoldingsRepo {
  return {
    replaceSnapshot(schemeId, asOfDate, rows) {
      db.delete(schemeHoldings)
        .where(and(eq(schemeHoldings.schemeId, schemeId), eq(schemeHoldings.asOfDate, asOfDate)))
        .run();
      for (const r of rows) {
        db.insert(schemeHoldings).values({
          schemeId, asOfDate,
          securityName: r.securityName, isin: r.isin,
          weightPct: r.weightPct, sector: r.sector, marketCapBucket: r.marketCapBucket,
        }).run();
      }
    },
    getLatestSnapshot(schemeId) {
      const latest = db.select({ d: sql<string>`MAX(${schemeHoldings.asOfDate})` })
        .from(schemeHoldings).where(eq(schemeHoldings.schemeId, schemeId)).get();
      if (!latest?.d) return [];
      return db.select().from(schemeHoldings)
        .where(and(eq(schemeHoldings.schemeId, schemeId), eq(schemeHoldings.asOfDate, latest.d)))
        .all() as StoredFundHolding[];
    },
    listAsOfDates(schemeId) {
      return db.selectDistinct({ d: schemeHoldings.asOfDate }).from(schemeHoldings)
        .where(eq(schemeHoldings.schemeId, schemeId))
        .orderBy(desc(schemeHoldings.asOfDate)).all().map((r) => r.d);
    },
    prune(schemeId, keep) {
      const dates = this.listAsOfDates(schemeId);
      const keepDates = dates.slice(0, keep);
      if (keepDates.length === dates.length) return 0;
      const res = db.delete(schemeHoldings)
        .where(and(eq(schemeHoldings.schemeId, schemeId), notInArray(schemeHoldings.asOfDate, keepDates)))
        .run();
      return res.changes;
    },
  };
}
```

import type { FundDataAdapter } from './types';
import { FetchedFundDataSchema } from './types';
import type { FetchedFundData, MarketCapBucket } from '../../types';

const GROWW_BASE = 'https://groww.in/mutual-funds';

function bucketFrom(raw: string | null | undefined): MarketCapBucket | null {
  if (!raw) return null;
  const s = raw.toLowerCase();
  if (s.includes('large')) return 'large';
  if (s.includes('mid')) return 'mid';
  if (s.includes('small')) return 'small';
  return 'other';
}

/** Pure — parse the __NEXT_DATA__ JSON object into normalized fund data. */
export function parseGrowwFundData(json: unknown, _amfiCode: string): FetchedFundData {
  const root = json as any;
  const pp = root?.props?.pageProps ?? {};
  // NOTE: key paths confirmed from packages/core/test/fixtures/fundData/groww-*.json (Task 1)
  const holdingsRaw: any[] = pp?.holdings ?? pp?.schemeHoldings?.holdings ?? [];
  const holdings = holdingsRaw.map((h) => ({
    securityName: String(h.companyName ?? h.name ?? '').trim(),
    isin: h.isin ?? null,
    weightPct: Number(h.corpusPer ?? h.holdingPercentage ?? h.weightage ?? 0),
    sector: h.sector ?? null,
    marketCapBucket: bucketFrom(h.marketCapType ?? h.marketCap),
  })).filter((h) => h.securityName.length > 0);

  const stats = pp?.analysis ?? pp?.stats ?? {};
  const result: FetchedFundData = {
    source: 'groww',
    asOfDate: String(pp?.holdingsAsOn ?? pp?.portfolioAsOn ?? new Date().toISOString().slice(0, 10)),
    fundamentals: {
      expenseRatioDirect: stats?.expenseRatio?.direct ?? pp?.expenseRatio ?? null,
      expenseRatioRegular: stats?.expenseRatio?.regular ?? null,
      planType: null,
      aum: pp?.aum ?? stats?.aum ?? null,
      benchmarkName: pp?.benchmarkName ?? null,
      stdDev: stats?.standardDeviation ?? null,
      sharpe: stats?.sharpeRatio ?? null,
      beta: stats?.beta ?? null,
      alpha: stats?.alpha ?? null,
      source: 'groww',
    },
    holdings,
  };
  return FetchedFundDataSchema.parse(result);
}

export const fetchGrowwFundData: FundDataAdapter = async (amfiCode, ctx) => {
  const slug = (ctx.schemeName ?? '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  const res = await fetch(`${GROWW_BASE}/${slug}`, { headers: { 'User-Agent': 'Mozilla/5.0' } });
  if (!res.ok) throw new Error(`Groww fetch failed: ${res.status}`);
  const html = await res.text();
  const m = html.match(/__NEXT_DATA__" type="application\/json">(.*?)<\/script>/s);
  if (!m) throw new Error('Groww: __NEXT_DATA__ not found');
  return parseGrowwFundData(JSON.parse(m[1]), amfiCode);
};
```

import { z } from 'zod';
import type { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import type { McpContext } from '../../context';
import { ok, errorResult, type ToolResult } from '../../shared/output';

export async function runRefreshFundData(
  ctx: McpContext, input: { amfiCode: string; source?: 'groww' | 'tickertape' | 'kuvera' },
): Promise<ToolResult> {
  try {
    const r = await ctx.fundData.ingest(input.amfiCode, {});
    return ok({ ...r });
  } catch (err) {
    return errorResult(`Fund-data refresh failed for ${input.amfiCode}: ${err instanceof Error ? err.message : String(err)}`);
  }
}

export function registerFundDataWriteTools(server: McpServer, ctx: McpContext): void {
  server.registerTool('refresh_fund_data', {
    description: 'Fetch and store the latest holdings + fundamentals for a mutual fund by AMFI code, ' +
      'from unofficial platform JSON. Required `amfiCode`; optional `source`. On source outage returns ' +
      'an error result (transient). Call this before analyzing when holdings are missing or stale.',
    inputSchema: { amfiCode: z.string().min(1), source: z.enum(['groww', 'tickertape', 'kuvera']).optional() },
  }, async (input) => runRefreshFundData(ctx, input));
}
```
